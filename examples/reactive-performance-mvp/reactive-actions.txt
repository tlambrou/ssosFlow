ACTION demo.reset
TRIGGER midi note ch=10 note=36
DO rhythm insert 0
DO rhythm route 0 density 1
DO rhythm route 0 chance 1
DO rhythm route 0 priority_mode 0
DO rhythm route 0 rotation 0
DO state groove reset
DO cue 0
END
